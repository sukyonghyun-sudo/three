const GradeShader = {
  uniforms:{ tDiffuse:{value:null}, res:{value:new THREE.Vector2(1,1)}, time:{value:0}, amt:{value:0.7}, mode:{value:0} },
  vertexShader:`varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
  fragmentShader:`
    uniform sampler2D tDiffuse; uniform vec2 res; uniform float time, amt; uniform int mode;
    varying vec2 vUv;
    float luma(vec3 c){ return dot(c, vec3(0.299,0.587,0.114)); }
    float hash(vec2 p){ return fract(sin(dot(p, vec2(127.1,311.7)))*43758.5453123); }
    vec3 post(vec3 c, float s){ return floor(c*s + 0.5)/s; }
    vec3 sat(vec3 c, float s){ return mix(vec3(luma(c)), c, s); }
    vec3 con(vec3 c, float k){ return clamp((c-0.5)*k + 0.5, 0.0, 1.0); }
    float edge(vec2 uv){
      vec2 t = 1.5/res;
      float tl=luma(texture2D(tDiffuse,uv+vec2(-t.x,t.y)).rgb), tc=luma(texture2D(tDiffuse,uv+vec2(0.0,t.y)).rgb);
      float tr=luma(texture2D(tDiffuse,uv+t).rgb), ml=luma(texture2D(tDiffuse,uv+vec2(-t.x,0.0)).rgb);
      float mr=luma(texture2D(tDiffuse,uv+vec2(t.x,0.0)).rgb), bl=luma(texture2D(tDiffuse,uv+vec2(-t.x,-t.y)).rgb);
      float bc=luma(texture2D(tDiffuse,uv+vec2(0.0,-t.y)).rgb), br=luma(texture2D(tDiffuse,uv+vec2(t.x,-t.y)).rgb);
      float gx = -tl -2.0*ml -bl +tr +2.0*mr +br, gy = tl +2.0*tc +tr -bl -2.0*bc -br;
      return clamp(sqrt(gx*gx + gy*gy), 0.0, 1.0);
    }
    void main(){
      vec2 uv = vUv;
      if (mode==6) uv += vec2(sin(uv.y*36.0 + time*1.8), 0.0)*0.0022*amt;     // 열기 아지랑이
      vec3 c = texture2D(tDiffuse, uv).rgb;
      float l = luma(c), g = hash(vUv*res + fract(time)*91.7) - 0.5;
      float vig = pow(length(vUv-0.5)*1.35, 2.0);
      vec3 o = c;
      if (mode==1){ o = con(sat(c, 1.06), 1.18) * (1.0 - 0.45*amt*vig); o = mix(c, o*vec3(0.97,1.0,1.06), amt); }
      else if (mode==2) o = mix(c, vec3(l)*vec3(1.16,0.95,0.72), amt);
      else if (mode==3){ o = mix(c, vec3(con(vec3(l),1.25).r) + g*0.12, amt); o *= 1.0 - 0.45*amt*vig; }
      else if (mode==4){ vec3 p = post(sat(c,1.35), mix(12.0,4.0,amt)); o = mix(p, vec3(0.05,0.04,0.07), smoothstep(0.25,0.55,edge(uv))*amt); }
      else if (mode==5){ o = mix(post(sat(c,1.4), 4.0), vec3(0.02), step(0.22, edge(uv))*amt); }
      else if (mode==6){ o = mix(c, sat(c*vec3(1.16,1.02,0.82), 1.12) + 0.04, amt); o *= 1.0 - 0.3*amt*vig; }
      else if (mode==7){ float sc = 0.72 + 0.28*sin(vUv.y*res.y*1.6); o = con(c*mix(1.0,sc,amt), 1.15) * (1.0 - 0.5*amt*vig); }
      else if (mode==8){ vec3 d = mix(vec3(0.15,0.02,0.30), vec3(1.0,0.35,0.85), l);
        d = mix(d, vec3(0.25,0.95,1.0), smoothstep(0.62,1.0,l));
        o = mix(c, d*(0.82+0.18*sin(vUv.y*res.y*1.4)), amt) + max(l-0.7,0.0)*vec3(1.0,0.4,0.9)*amt; }
      else if (mode==9){ o = mix(c, vec3(con(vec3(l),1.9).r) + g*0.09, amt); o *= 1.0 - 0.6*amt*vig; }
      gl_FragColor = vec4(clamp(o,0.0,1.0), 1.0);
    }`

const norm = n => n.toLowerCase().replace(/^.*:/, '').replace(/mixamorig/g, '').replace(/[^a-z0-9]/g, '');
let boneMap = new Map();
function retarget(clip){
  const tracks = [];
  for (const t of clip.tracks){
    const dot = t.name.lastIndexOf('.');
    if (dot < 0) continue;
    const node = norm(t.name.slice(0, dot));
    const target = boneMap.get(node);
    if (!target) continue;
    const nt = t.clone(); nt.name = target + t.name.slice(dot);
    // 힙의 수평 이동은 제거 → 제자리 모션 (이동은 키보드가 담당)
    if (node === 'hips' && t.name.endsWith('.position')){
      const v = nt.values, x0 = v[0], z0 = v[2];
      for (let k = 0; k < v.length; k += 3){ v[k] -= x0; v[k+2] -= z0; }
    }
    tracks.push(nt);
  }
  return new THREE.AnimationClip(clip.name, clip.duration, tracks);
}

function prepMeshes(root){
  root.traverse(o => {
    if (o.isMesh || o.isSkinnedMesh){
      o.castShadow = true; o.receiveShadow = true; o.frustumCulled = false;
      const mats = Array.isArray(o.material) ? o.material : [o.material];
      mats.forEach(m => {
        if (!m) return;
        if (m.map) m.map.colorSpace = THREE.SRGBColorSpace;
        m.side = THREE.DoubleSide; m.needsUpdate = true;
      });
    }
  });
}

/* 접지 보정을 실측한다.
   FBX 바인드 포즈의 바운딩 박스는 원점이 발이 아니라 몸 중앙인 경우가 많아
   -box.min.y 로 올리면 캐릭터가 키의 절반만큼 공중에 뜬다.
   애니메이션이 적용된 실제 발 본 높이로 잡는 게 정확하다. */
function calibrateGround(){
  if (!character || !mixer) return;
  mixer.update(0);
  character.position.y = 0;
  character.updateMatrixWorld(true);
  const p = new THREE.Vector3();
  let low = Infinity;
  ['lefttoebase','righttoebase','leftfoot','rightfoot'].forEach(k => {
    const b = findBone(character, k);
    if (b){ b.getWorldPosition(p); low = Math.min(low, p.y); }
  });
  if (low === Infinity){                       // 발 본이 없으면 메시 최저점으로
    const bb = new THREE.Box3().setFromObject(character);
    low = bb.min.y;
  }
  charGroundY = -low;
  character.position.y = charGroundY;
  if (dummy) dummy.position.y = 0;
}

function findHandBone(root){
  let hit = null;
  root.traverse(o => {
    if (hit || !o.isBone) return;
    const n = norm(o.name);
    if (n.includes('righthand') && !/(index|thumb|middle|ring|pinky|end)/.test(n)) hit = o;
  });
  if (!hit) root.traverse(o => { if (!hit && o.isBone && /r.*hand/i.test(o.name)) hit = o; });
  return hit;
}

async function getClip(m){
  if (clipCache.has(m.name)) return clipCache.get(m.name);
  const { obj } = await loadFirst(m.files);
  const raw = obj.animations && obj.animations[0];
  if (!raw) throw new Error('no clip');
  const clip = retarget(raw);
  clipCache.set(m.name, clip);
  return clip;
}

// 클립 전환은 동기로 처리한다. 비동기로 한 프레임이라도 비면 그 사이 바인드(T) 포즈가 보인다.
function startClip(i, clip, fade){
  const m = MOTIONS[i];
  const once = m.loop === false;
  const action = mixer.clipAction(clip);
  action.reset();
  action.setLoop(once ? THREE.LoopOnce : THREE.LoopRepeat, once ? 1 : Infinity);
  action.clampWhenFinished = true;          // 끝나도 마지막 포즈를 유지 → T포즈로 튀지 않음
  action.paused = !playing;
  action.play();
  if (currentAction && currentAction !== action){
    currentAction.crossFadeTo(action, fade, false);
    const prev = currentAction;
    setTimeout(() => { if (prev !== currentAction) prev.stop(); }, fade*1000 + 80);
  }
  currentAction = action; currentClip = clip; activeIndex = i;
  buttons.forEach((b, n) => b.setAttribute('aria-pressed', String(n === i)));
  if (!controlMode || i > IDX_RUN) setStatus(m.name + ' · ' + clip.duration.toFixed(2) + '초');
}

async function selectMotion(i, fade = 0.18){
  if (!mixer || pendingIndex === i) return;
  const m = MOTIONS[i];
  if (activeIndex === i){
    if (m.loop === false && currentAction){ currentAction.reset().play(); currentAction.paused = !playing; }
    return;
  }
  const cached = clipCache.get(m.name);
  if (cached){ startClip(i, cached, fade); return; }

  buttons[i].querySelector('.tag').textContent = '…';
  pendingIndex = i;
  try {
    const clip = await getClip(m);
    pendingIndex = -1;
    buttons[i].querySelector('.tag').textContent = '';
    startClip(i, clip, fade);
  } catch {
    pendingIndex = -1;
    buttons[i].querySelector('.tag').textContent = '없음';
    setStatus(m.files[0] + ' 을(를) 불러오지 못했습니다.', true);
  }
}

// 같은 attribute 를 공유하고 인덱스만 다른 지오메트리 (추가 메모리 거의 없음)
function sharedGeo(src, indexArray){
  const g = new THREE.BufferGeometry();
  for (const k in src.attributes) g.setAttribute(k, src.attributes[k]);

// 잘린 본 무리에 실린 정점을 골라 삼각형을 두 덩어리로 나눈다
function splitGeometry(mesh, severed){
  const geo = mesh.geometry;
  const si = geo.attributes.skinIndex, sw = geo.attributes.skinWeight;
  const n = geo.attributes.position.count;
  const flag = new Uint8Array(n);
  for (let v = 0; v < n; v++){
    let s = 0;
    if (severed.has(si.getX(v))) s += sw.getX(v);
    if (severed.has(si.getY(v))) s += sw.getY(v);
    if (severed.has(si.getZ(v))) s += sw.getZ(v);
    if (severed.has(si.getW(v))) s += sw.getW(v);
    flag[v] = s > 0.5 ? 1 : 0;
  }
  const idx = geo.index ? geo.index.array : null;
  const total = idx ? idx.length : n;
  const body = [], part = [];
  for (let t = 0; t + 2 < total; t += 3){
    const a = idx ? idx[t] : t, b = idx ? idx[t+1] : t+1, c = idx ? idx[t+2] : t+2;
    ((flag[a] + flag[b] + flag[c]) >= 2 ? part : body).push(a, b, c);
  }
  return { body: new Uint32Array(body), part: new Uint32Array(part) };
}

// 절단 경계(남는 쪽/떨어지는 쪽이 공유하는 변)를 찾아 부채꼴로 메운 스킨 메시를 만든다
const capMat = new THREE.MeshStandardMaterial({ color:0x7d161c, roughness:0.95, side:THREE.DoubleSide });
function buildCapGeometry(srcGeo, partIdx, bodyIdx, cutBoneIndex){
  const pos = srcGeo.attributes.position, si = srcGeo.attributes.skinIndex, sw = srcGeo.attributes.skinWeight;
  const N = pos.count;
  // 비인덱스 지오메트리는 정점이 삼각형마다 복제돼 있어 위치로 용접한다
  const canon = new Map(); const cidCache = new Int32Array(N).fill(-1);
  const cid = i => {
    if (cidCache[i] >= 0) return cidCache[i];
    const k = (Math.round(pos.getX(i)*1e3)) + ',' + (Math.round(pos.getY(i)*1e3)) + ',' + (Math.round(pos.getZ(i)*1e3));
    let r = canon.get(k); if (r === undefined){ canon.set(k, i); r = i; }
    cidCache[i] = r; return r;
  };
  const ek = (a, b) => a < b ? a*N + b : b*N + a;
  const bodyEdges = new Set();
  for (let t = 0; t < bodyIdx.length; t += 3){
    const a = cid(bodyIdx[t]), b2 = cid(bodyIdx[t+1]), c = cid(bodyIdx[t+2]);
    bodyEdges.add(ek(a,b2)); bodyEdges.add(ek(b2,c)); bodyEdges.add(ek(c,a));
  }
  const adj = new Map();
  const link = (a, b2) => { if (!adj.has(a)) adj.set(a, []); if (!adj.has(b2)) adj.set(b2, []); adj.get(a).push(b2); adj.get(b2).push(a); };
  const seen = new Set();
  for (let t = 0; t < partIdx.length; t += 3){
    const v = [cid(partIdx[t]), cid(partIdx[t+1]), cid(partIdx[t+2])];
    for (let e = 0; e < 3; e++){
      const a = v[e], b2 = v[(e+1)%3], k = ek(a,b2);
      if (bodyEdges.has(k) && !seen.has(k)){ seen.add(k); link(a, b2); }
    }
  }
  if (!adj.size) return null;

  // 경계 변을 따라 고리(loop)로 묶는다
  const visited = new Set(), loops = [];
  for (const start of adj.keys()){
    if (visited.has(start)) continue;
    const loop = []; let cur = start, prev = -1;
    while (cur !== undefined && !visited.has(cur)){
      visited.add(cur); loop.push(cur);
      const nb = adj.get(cur).filter(n => n !== prev && !visited.has(n));
      prev = cur; cur = nb[0];
    }
    if (loop.length >= 3) loops.push(loop);
  }
  if (!loops.length) return null;

  const P = [], SI = [], SW = [], IDX = [];
  let base = 0;
  for (const loop of loops){
    const cx = [0,0,0];
    for (const i of loop){ cx[0] += pos.getX(i); cx[1] += pos.getY(i); cx[2] += pos.getZ(i); }
    for (let k = 0; k < 3; k++) cx[k] /= loop.length;
    for (const i of loop){
      P.push(pos.getX(i), pos.getY(i), pos.getZ(i));
      SI.push(si.getX(i), si.getY(i), si.getZ(i), si.getW(i));
      SW.push(sw.getX(i), sw.getY(i), sw.getZ(i), sw.getW(i));
    }
    P.push(cx[0], cx[1], cx[2]); SI.push(cutBoneIndex, 0, 0, 0); SW.push(1, 0, 0, 0);
    const c = base + loop.length;
    for (let k = 0; k < loop.length; k++) IDX.push(base + k, base + (k+1) % loop.length, c);
    base += loop.length + 1;
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(P, 3));
  g.setAttribute('skinIndex', new THREE.Uint16BufferAttribute(SI, 4));
  g.setAttribute('skinWeight', new THREE.Float32BufferAttribute(SW, 4));
  g.setIndex(IDX);
  g.computeVertexNormals();
  g.boundingSphere = new THREE.Sphere(new THREE.Vector3(), 1e4);
  return g;
}
function attachCap(capGeo, refMesh){
  const m = new THREE.SkinnedMesh(capGeo, capMat);
  m.position.copy(refMesh.position); m.quaternion.copy(refMesh.quaternion); m.scale.copy(refMesh.scale);
  m.frustumCulled = false; m.castShadow = true;
  refMesh.parent.add(m);
  m.bind(refMesh.skeleton, refMesh.bindMatrix);
  return m;
}

function findBone(root, key){
  let hit = null;
  root.traverse(o => { if (!hit && o.isBone && norm(o.name) === key) hit = o; });
  if (!hit) root.traverse(o => { if (!hit && o.isBone && norm(o.name).indexOf(key) === 0) hit = o; });

function cut(index){
  if (!dummy) return;
  const def = CUTS[index];
  const bone = findBone(dummy, def.bone);
  if (!bone || cutBones.has(bone)) return;

  const wp = new THREE.Vector3(); bone.getWorldPosition(wp);
  const dir = new THREE.Vector3(def.dir[0], def.dir[1], def.dir[2]).normalize();
  const dSkins = skinsOf(dummy);
  if (!dSkins.length) return;

  // 1) 각 스킨 메시를 남는 쪽 / 떨어지는 쪽 삼각형으로 나눈다
  const splits = dSkins.map(sm => {
    if (!geoBackup.has(sm.uuid)) geoBackup.set(sm.uuid, sm.geometry);
    const s = splitGeometry(sm, subtreeBoneIndices(sm.skeleton, bone));
    const bi = sm.skeleton.bones.indexOf(bone);
    s.cap = s.part.length ? buildCapGeometry(geoBackup.get(sm.uuid), s.part, s.body, Math.max(bi, 0)) : null;
    return s;
  });

  // 2) 떨어져 나갈 조각: 현재 포즈로 복제한 뒤 '떨어지는 쪽' 삼각형만 남긴다
  let piece = null;
  if (blood.piece && splits.some(s => s.part.length)){
    const clone = SkeletonUtils.clone(dummy);
    prepMeshes(clone);
    const cSkins = skinsOf(clone);
    cSkins.forEach((cm, i) => {
      cm.geometry = sharedGeo(geoBackup.get(dSkins[i].uuid), splits[i].part);
      if (splits[i].cap) attachCap(splits[i].cap, cm);
    });

    const keepRoot = findBone(clone, norm(bone.name));
    const keep = new Set(); if (keepRoot) keepRoot.traverse(o => { if (o.isBone) keep.add(o); });

    piece = new THREE.Group();
    clone.position.copy(dummy.position).sub(wp);
    clone.quaternion.copy(dummy.quaternion);
    clone.scale.copy(dummy.scale);
    piece.add(clone);
    piece.updateMatrixWorld(true);

    // 남은 관절 위치로 조각의 바운딩 박스를 잡는다 (지오메트리 박스는 T포즈 전신이라 못 쓴다)
    const box = new THREE.Box3(); const v = new THREE.Vector3();
    keep.forEach(b => { b.getWorldPosition(v); box.expandByPoint(v); });
    if (box.isEmpty()) box.setFromCenterAndSize(wp, new THREE.Vector3(0.2,0.2,0.2));
    box.expandByScalar(def.margin);
    const corners = [];
    for (let k = 0; k < 8; k++) corners.push(new THREE.Vector3((k&1)?box.max.x:box.min.x, (k&2)?box.max.y:box.min.y, (k&4)?box.max.z:box.min.z));
    piece.userData.corners = corners;
    piece.userData.radius = box.getSize(new THREE.Vector3()).length()*0.5;
    piece.position.copy(wp);

    scene.add(piece);
    const d = dir.clone().applyQuaternion(dummy.quaternion);
    pieces.push({ mesh:piece,
      vel: d.multiplyScalar(1.0 + Math.random()*0.9).add(new THREE.Vector3(0, 1.4, 0)),
      spin: new THREE.Vector3((Math.random()-0.5)*5, (Math.random()-0.5)*5, (Math.random()-0.5)*5),
      rest:false });
  }

  // 3) 본체에서 그 부위의 삼각형을 제거한다 → 실제로 떨어져 나간다
  dSkins.forEach((sm, i) => {
    sm.geometry = sharedGeo(geoBackup.get(sm.uuid), splits[i].body);
    if (splits[i].cap) bodyCaps.push(attachCap(splits[i].cap, sm));
  });
  cutBones.add(bone);

  // 4) 분출
  const boost = (typeof cine !== 'undefined' && cine.active) ? 2.2 : 1;   // 연출 중에는 더 많이
  emitters.push({ bone, dir, t:0, dur:blood.dur*(boost > 1 ? 1.8 : 1), rate:blood.amount*boost });
  const dWorld = dir.clone().applyQuaternion(dummy.quaternion);
  for (let i = 0, n = Math.round(blood.amount*0.35*boost); i < n; i++) emitOne(wp, dWorld, 1.6);

  dummyReact(def.kind !== 'limb');
  setStatus('더미 ' + def.name + ' 절단');
}

const _wp = new THREE.Vector3(), _cv = new THREE.Vector3(), _axis = new THREE.Vector3(), _qq = new THREE.Quaternion();
function updateGore(dt){
  for (let i = emitters.length-1; i >= 0; i--){
    const e = emitters[i]; e.t += dt;
    if (e.t > e.dur){ emitters.splice(i,1); continue; }
    e.bone.getWorldPosition(_wp);
    const fade = 1 - e.t/e.dur;
    const n = Math.max(1, Math.round(e.rate * dt * 1.4 * fade));
    const d = e.dir.clone().applyQuaternion(dummy.quaternion);
    for (let k = 0; k < n; k++) emitOne(_wp, d, 0.6 + fade);
  }
  let dirty = false;
  for (let i = 0; i < MAXP; i++){
    if (pLife[i] <= 0) continue;
    pLife[i] -= dt; const i3 = i*3;
    pVel[i3+1] -= blood.grav*dt;
    pPos[i3] += pVel[i3]*dt; pPos[i3+1] += pVel[i3+1]*dt; pPos[i3+2] += pVel[i3+2]*dt;
    if (pPos[i3+1] <= terrainY(pPos[i3], pPos[i3+2]) + 0.01){
      drawSplat(pPos[i3], pPos[i3+2], pSize[i]*7 + 0.02, 0.55);
      pLife[i] = 0; pAlpha[i] = 0; pPos[i3+1] = -99;
    }
    else { pAlpha[i] = pLife[i] < 0.35 ? pLife[i]/0.35 : 1; if (pLife[i] <= 0){ pAlpha[i] = 0; pPos[i3+1] = -99; } }
    dirty = true;
  }
  if (dirty){ pGeo.attributes.position.needsUpdate = true; pGeo.attributes.aAlpha.needsUpdate = true; pGeo.attributes.aSize.needsUpdate = true; }

  // 조각 물리: 관절 기반 바운딩 박스로 바닥 충돌 → 넘어지고 구르다 멈춘다
  for (let i = pieces.length-1; i >= 0; i--){
    const p = pieces[i];
    if (p.rest) continue;
    p.vel.y -= 12.5*dt;
    p.mesh.position.addScaledVector(p.vel, dt);
    const w = p.spin.length();
    if (w > 1e-4){ _axis.copy(p.spin).multiplyScalar(1/w); _qq.setFromAxisAngle(_axis, w*dt); p.mesh.quaternion.premultiply(_qq); }
    p.mesh.updateMatrix();

    let minY = Infinity, lx = 0, lz = 0;
    for (const c of p.mesh.userData.corners){
      _cv.copy(c).applyMatrix4(p.mesh.matrix);
      if (_cv.y < minY){ minY = _cv.y; lx = _cv.x; lz = _cv.z; }
    }
    const gY = terrainY(lx, lz);
    if (minY < gY){
      p.mesh.position.y += gY - minY;
      const impact = Math.max(-p.vel.y, 0);
      drawSplat(lx, lz, 0.18 + p.mesh.userData.radius*0.4, 0.35);
      p.spin.x += (lz - p.mesh.position.z) * impact * 2.2;
      p.spin.z -= (lx - p.mesh.position.x) * impact * 2.2;
      p.vel.y = impact > 0.9 ? impact*0.28 : 0;
      p.vel.x *= 0.7; p.vel.z *= 0.7; p.spin.multiplyScalar(0.6);
      if (p.vel.length() + p.spin.length()*0.2 < 0.4){
        p.rest = true; p.vel.set(0,0,0); p.spin.set(0,0,0);
        drawSplat(p.mesh.position.x, p.mesh.position.z, 0.25 + p.mesh.userData.radius, 0.5);
      }
    } else p.spin.multiplyScalar(1 - Math.min(0.3*dt, 0.5));
  }
}

// 두 선분 사이 최단거리 (Ericson, Real-Time Collision Detection)
function segSegDist(p1, q1, p2, q2){
  _d1.subVectors(q1, p1); _d2.subVectors(q2, p2); _r.subVectors(p1, p2);
  const a = _d1.dot(_d1), e = _d2.dot(_d2), f = _d2.dot(_r);
  let s, t;
  if (a <= 1e-9 && e <= 1e-9){ return _r.length(); }
  if (a <= 1e-9){ s = 0; t = THREE.MathUtils.clamp(f/e, 0, 1); }
  else {
    const c = _d1.dot(_r);
    if (e <= 1e-9){ t = 0; s = THREE.MathUtils.clamp(-c/a, 0, 1); }
    else {
      const b = _d1.dot(_d2), den = a*e - b*b;
      s = den !== 0 ? THREE.MathUtils.clamp((b*f - c*e)/den, 0, 1) : 0;
      t = (b*s + f)/e;
      if (t < 0){ t = 0; s = THREE.MathUtils.clamp(-c/a, 0, 1); }
      else if (t > 1){ t = 1; s = THREE.MathUtils.clamp((b - c)/a, 0, 1); }
    }
  }
  _c1.copy(p1).addScaledVector(_d1, s); _c2.copy(p2).addScaledVector(_d2, t);
  return _c1.distanceTo(_c2);
}
function bladeNow(base, tip){
  holder.getWorldPosition(base);
  _fw.set(0,1,0).applyQuaternion(holder.getWorldQuaternion(_q));
  tip.copy(base).addScaledVector(_fw, T.len);
}
let partBones = null;
function cachePartBones(){
  if (!dummy) return;
  partBones = PARTS.map(p => ({ a: findBone(dummy, p.a), b: findBone(dummy, p.b) }));

function partSegment(i, a, b){
  if (!partBones) cachePartBones();
  const pb = partBones && partBones[i];
  if (!pb || !pb.a) return false;
  pb.a.getWorldPosition(a);
  if (pb.b) pb.b.getWorldPosition(b); else b.copy(a).setY(a.y + 0.2);
  return true;
}
const SUB = 5;                         // 프레임 사이를 몇 단면으로 쪼갤지

function bladeSweep(){
  if (!holder || !dummy || !dummy.visible || !hitCut) return;
  bladeNow(_b0, _b1);
  const done = new Set(); cutBones.forEach(bn => done.add(norm(bn.name)));
  let best = null, bestD = Infinity;

  for (let i = 0; i < PARTS.length; i++){
    const part = PARTS[i];
    if (done.has(CUTS[part.cut].bone)) continue;
    if (!partSegment(i, _pa, _pb)) continue;
    let d = segSegDist(_b0, _b1, _pa, _pb);
    if (prevBlade.valid){
      // 이전 칼날 → 현재 칼날 사이를 보간해 빠른 스윙이 프레임을 건너뛰어도 잡는다
      for (let s = 1; s < SUB; s++){
        const t = s/SUB;
        _s0.lerpVectors(prevBlade.base, _b0, t);
        _s1.lerpVectors(prevBlade.tip, _b1, t);
        const ds = segSegDist(_s0, _s1, _pa, _pb);
        if (ds < d) d = ds;
      }
    }
    if (d < bestD){ bestD = d; if (d < swingMin){ swingMin = d; swingPart = part; } }
    if (d < part.r + hitR && d < (best ? best._d : Infinity)){ best = part; best._d = d; }
  }
  prevBlade.base.copy(_b0); prevBlade.tip.copy(_b1); prevBlade.valid = true;

  if (best){
    hitDone = true;
    cut(best.cut);
    hitStop = hitStopLen;
    const dx = dummy.position.x - character.position.x, dz = dummy.position.z - character.position.z, dl = Math.hypot(dx, dz) || 1;
    dummy.position.x += dx/dl*0.06; dummy.position.z += dz/dl*0.06;
  }
}

// 실시간으로 카메라·타임라인을 돌리고, 월드에 적용할 시간 배율을 돌려준다
function updateCinematic(rdt){
  cine.t += rdt;
  const t = cine.t;

  if (cine.action){
    // 접촉 시점(CUT_AT)에 클립이 정확히 55% 가 되도록 나눠 매핑한다
    let p;
    if (t < cine.CUT_AT){
      p = 0.10 + 0.45*ease(THREE.MathUtils.clamp((t - 0.15)/(cine.CUT_AT - 0.15), 0, 1));
    } else {
      p = 0.55 + 0.35*ease(THREE.MathUtils.clamp((t - cine.CUT_AT)/1.5, 0, 1));
    }
    cine.action.time = cine.dur * p;
  }
  if (!cine.cutDone && t >= cine.CUT_AT){
    cine.cutDone = true;
    cut(cine.part);
    cineLabel.classList.add('on');
    setTimeout(() => cineLabel.classList.remove('on'), 900);
  }

  const K = cine.K, C = cine.CUT_AT;
  let a, b, u;
  if (t < C){ a = K[0]; b = K[1]; u = t/C; }                                   // 스윙 전체를 보여주는 구간
  else if (t < C + 1.3){ a = K[1]; b = K[2]; u = (t - C)/1.3; }                // 갈라지는 동안 반대편으로
  else { a = K[2]; b = K[3]; u = (t - C - 1.3)/(cine.TOTAL - C - 1.3); }       // 얼굴 리빌 + 달리 아웃
  u = ease(THREE.MathUtils.clamp(u, 0, 1));

  _kp.lerpVectors(a.pivot, b.pivot, u);
  _kt.lerpVectors(a.target, b.target, u);
  if (cine.head && t > C + 0.5){ cine.head.getWorldPosition(_tmp); _kt.lerp(_tmp, 0.9); }

  const h  = THREE.MathUtils.lerp(a.h, b.h, u);
  const az = THREE.MathUtils.degToRad(THREE.MathUtils.lerp(a.az, b.az, u));
  const el = THREE.MathUtils.degToRad(THREE.MathUtils.lerp(a.el, b.el, u));

  // 담을 세로 크기(h)를 화각에 맞춰 거리로 환산 — 이게 없으면 망원에서 프레임이 잘린다
  const vfov = THREE.MathUtils.degToRad(camera.fov);
  const aspect = Math.max(camera.aspect, 0.1);
  let d = (h*0.5) / Math.tan(vfov*0.5) * 1.12;
  if (aspect < 1) d /= aspect;                                  // 세로 화면에서는 더 물러난다
  d = Math.max(d, 1.2);

  const bx = -cine.F.x, bz = -cine.F.z;
  const rx = bx*Math.cos(az) - bz*Math.sin(az);
  const rz = bx*Math.sin(az) + bz*Math.cos(az);
  const hz = Math.cos(el);
  camera.position.set(_kp.x + rx*hz*d, _kp.y + Math.sin(el)*d, _kp.z + rz*hz*d);
  if (camera.position.y < 0.25) camera.position.y = 0.25;        // 바닥 아래로 내려가지 않게
  camera.lookAt(_kt);

  if (t >= cine.TOTAL){ endCinematic(); return 1; }
  if (t < cine.CUT_AT - 0.35) return Math.max(cine.slow*1.6, 0.18);   // 접근·윈드업
  if (t < cine.CUT_AT + 1.5) return cine.slow;                   // 가르는 순간
  return THREE.MathUtils.lerp(cine.slow, 1, THREE.MathUtils.clamp((t - cine.CUT_AT - 1.5)/0.9, 0, 1));
}

function updateControl(dt){
  if (!character || !controlMode || attacking) return;
  const f = (keys['KeyW']||keys['ArrowUp']?1:0) - (keys['KeyS']||keys['ArrowDown']?1:0);
  const r = (keys['KeyD']||keys['ArrowRight']?1:0) - (keys['KeyA']||keys['ArrowLeft']?1:0);
  if (f || r){
    camera.getWorldDirection(_cf); _cf.y = 0; _cf.normalize();     // 화면 안쪽
    _cr.crossVectors(_cf, UP).normalize();                           // 화면 오른쪽
    let mx = _cf.x*f + _cr.x*r, mz = _cf.z*f + _cr.z*r;
    const len = Math.hypot(mx, mz) || 1; mx /= len; mz /= len;
    character.position.x += mx*moveSpeed*dt;
    character.position.z += mz*moveSpeed*dt;
    const yaw = Math.atan2(mx, mz) - (faceFlip ? Math.PI : 0);
    _q.setFromAxisAngle(UP, yaw);
    character.quaternion.slerp(_q, Math.min(1, dt*10));
    if (activeIndex !== IDX_RUN) selectMotion(IDX_RUN);
  } else if (activeIndex !== IDX_IDLE) selectMotion(IDX_IDLE);
}

function findParts(root){
  let n = root;
  while (n.children.length === 1) n = n.children[0];
  const parts = n.children.filter(ch => { let has = false; ch.traverse(o => { if (o.isMesh) has = true; }); return has; });
  return parts.length ? parts : [n];
}

/* 높이맵: 삼각형을 격자에 래스터라이즈해 굽는다.
   레이캐스트는 삼각형 전수 검사라 굽는 데만 수십 초가 걸린다. */
let hm = null, hmT = { x:0, y:0, z:0, s:1, rot:0 };
function bakeHeightmap(){
  hm = null;
  const meshes = [];
  if (mapObj) mapObj.traverse(o => { if (o.isMesh && o.visible && o.parent.visible) meshes.push(o); });
  if (!meshes.length){ reshapeSplat(); return; }

  const t0 = performance.now();
  mapObj.updateMatrixWorld(true);            // 부모 행렬이 stale 이면 직전 배치 기준으로 구워진다
  const box = new THREE.Box3();
  meshes.forEach(m => box.expandByObject(m));
  const N = 200;
  const minX = box.min.x, minZ = box.min.z;
  const sx = Math.max(box.max.x - minX, 1e-3), sz = Math.max(box.max.z - minZ, 1e-3);
  const data = new Float32Array(N*N).fill(-Infinity);
  const v = new THREE.Vector3();

  for (const m of meshes){
    m.updateMatrixWorld(true);
    const g = m.geometry, pos = g.attributes.position, idx = g.index;
    const wp = new Float32Array(pos.count*3);
    for (let i = 0; i < pos.count; i++){
      v.fromBufferAttribute(pos, i).applyMatrix4(m.matrixWorld);
      wp[i*3] = v.x; wp[i*3+1] = v.y; wp[i*3+2] = v.z;
    }
    const tri = idx ? idx.count : pos.count;
    for (let t = 0; t < tri; t += 3){
      const a = idx ? idx.getX(t) : t, b = idx ? idx.getX(t+1) : t+1, cc = idx ? idx.getX(t+2) : t+2;
      const ax = wp[a*3], ay = wp[a*3+1], az = wp[a*3+2];
      const bx = wp[b*3], by = wp[b*3+1], bz = wp[b*3+2];
      const cx = wp[cc*3], cy = wp[cc*3+1], cz = wp[cc*3+2];
      const gx0 = Math.max(0, Math.floor((Math.min(ax,bx,cx) - minX)/sx*(N-1)));
      const gx1 = Math.min(N-1, Math.ceil((Math.max(ax,bx,cx) - minX)/sx*(N-1)));
      const gz0 = Math.max(0, Math.floor((Math.min(az,bz,cz) - minZ)/sz*(N-1)));
      const gz1 = Math.min(N-1, Math.ceil((Math.max(az,bz,cz) - minZ)/sz*(N-1)));
      const d = (bz - cz)*(ax - cx) + (cx - bx)*(az - cz);
      if (Math.abs(d) < 1e-12) continue;
      for (let gz = gz0; gz <= gz1; gz++){
        const pz = minZ + gz/(N-1)*sz;
        for (let gx = gx0; gx <= gx1; gx++){
          const px = minX + gx/(N-1)*sx;
          const w0 = ((bz - cz)*(px - cx) + (cx - bx)*(pz - cz))/d;
          const w1 = ((cz - az)*(px - cx) + (ax - cx)*(pz - cz))/d;
          const w2 = 1 - w0 - w1;
          if (w0 < -0.001 || w1 < -0.001 || w2 < -0.001) continue;
          const y = w0*ay + w1*by + w2*cy;
          const k = gz*N + gx;
          if (y > data[k]) data[k] = y;
        }
      }
    }
  }
  hm = { n:N, minX, minZ, sx, sz, data };
  hmT = { x:mapT.x, y:mapT.y, z:mapT.z, s:mapT.s, rot:mapT.rot };
  hm.bias = 0;
  splatMesh.position.set(0, 0.004, 0);

  // 검산: 격자 중앙에 레이 한 번 쏴서 실제 표면과 비교하고 차이를 보정한다
  const cxw = minX + sx*0.5, czw = minZ + sz*0.5;
  mapRay.set(_from.set(cxw, box.max.y + 50, czw), _down);
  mapRay.far = box.max.y - box.min.y + 200;
  const hit = mapRay.intersectObjects(meshes, false)[0];
  let note = '';
  if (hit){
    const diff = hit.point.y - terrainY(cxw, czw);
    if (Math.abs(diff) > 0.03){ hm.bias = diff; note = ' · 보정 ' + diff.toFixed(2) + 'm'; }
  }
  if (placeWanted) placeOnTerrain();
  reshapeSplat();
  if (!placeWanted) setStatus(MAPS[mapIndex].name + ' · 높이맵 ' + N + '×' + N + ' · ' + Math.round(performance.now()-t0) + 'ms' + note);
}
function terrainY(x, z){
  if (!hm) return 0;
  // 평행이동은 격자를 옮기고 높이를 더하는 것으로 끝난다 → 다시 구울 필요가 없다
  const qx = x - (mapT.x - hmT.x), qz = z - (mapT.z - hmT.z), dy = mapT.y - hmT.y;
  const fx = (qx - hm.minX)/hm.sx*(hm.n-1), fz = (qz - hm.minZ)/hm.sz*(hm.n-1);
  if (fx < 0 || fz < 0 || fx > hm.n-1 || fz > hm.n-1) return 0;   // 지형 밖은 기준 평면
  const x0 = Math.floor(fx), z0 = Math.floor(fz);
  const x1 = Math.min(x0+1, hm.n-1), z1 = Math.min(z0+1, hm.n-1);
  const tx = fx - x0, tz = fz - z0;
  const g = (i, j) => { const val = hm.data[j*hm.n + i]; return val === -Infinity ? 0 : val; };
  return (g(x0,z0)*(1-tx) + g(x1,z0)*tx)*(1-tz) + (g(x0,z1)*(1-tx) + g(x1,z1)*tx)*tz + dy + (hm.bias || 0);
}
// 핏자국 평면을 지형에 덮어씌운다 (평면 그대로 두면 땅에 파묻힌다)
function reshapeSplat(){
  const seg = hm ? 72 : 1;
  splatMesh.geometry.dispose();
  const g = new THREE.PlaneGeometry(SPLAT_SIZE, SPLAT_SIZE, seg, seg);
  if (hm){
    const p = g.attributes.position;
    for (let i = 0; i < p.count; i++) p.setZ(i, terrainY(p.getX(i), -p.getY(i)) + 0.03);
    p.needsUpdate = true; g.computeVertexNormals();
  }
  splatMesh.geometry = g;
}

/* 굽힌 높이맵에서 평평한 자리를 찾아 캐릭터와 더미를 올려놓는다 */
let placeWanted = false;
function placeOnTerrain(){
  if (!character){ placeWanted = true; return; }
  if (!hm){
    character.position.set(0, charGroundY, 0);
    if (dummy) dummy.position.set(1.3, 0, 0);
    controls.target.set(0, 0.95, 0);
    camera.position.set(1.9, 1.5, 2.7);
    return;
  }
  const N = hm.n, lo = Math.floor(N*0.25), hi = Math.floor(N*0.75);
  let best = null, bestScore = Infinity;
  const at = (i, j) => { const v = hm.data[j*N + i]; return v === -Infinity ? null : v; };
  for (let j = lo; j < hi; j++){
    for (let i = lo; i < hi; i++){
      const h0 = at(i, j); if (h0 === null) continue;
      let mn = h0, mx = h0, ok = true;
      for (let dj = -2; dj <= 2 && ok; dj++)
        for (let di = -2; di <= 2; di++){
          const v = at(i+di, j+dj);
          if (v === null){ ok = false; break; }
          if (v < mn) mn = v; if (v > mx) mx = v;
        }
      if (!ok) continue;
      // 평탄할수록, 중앙에 가까울수록 좋다
      const cx = (i - N/2)/N, cz = (j - N/2)/N;
      const score = (mx - mn)*3 + Math.hypot(cx, cz);
      if (score < bestScore){ bestScore = score; best = [i, j, h0]; }
    }
  }
  if (!best) best = [Math.floor(N/2), Math.floor(N/2), 0];
  const wx = hm.minX + best[0]/(N-1)*hm.sx + (mapT.x - hmT.x);
  const wz = hm.minZ + best[1]/(N-1)*hm.sz + (mapT.z - hmT.z);

  const groundAt = (x, z) => {
    if (mapObj){
      mapRay.set(_from.set(x, 500, z), _down); mapRay.far = 2000;
      const h = mapRay.intersectObject(mapObj, true).find(r => r.object.visible && r.object.parent.visible);
      if (h) return h.point.y;
    }
    return terrainY(x, z);
  };
  const gy = groundAt(wx, wz);
  character.position.set(wx, gy + charGroundY, wz);
  character.quaternion.setFromAxisAngle(UP, faceFlip ? Math.PI : 0);
  if (dummy){
    const dx = wx, dz = wz + 1.4;
    dummy.position.set(dx, groundAt(dx, dz), dz);
    const el = document.getElementById('dumX');
    el.value = dx; el.dispatchEvent(new Event('input', { bubbles:true }));
  }
  controls.target.set(wx, character.position.y + 0.95, wz);
  camera.position.set(wx + 2.2, character.position.y + 1.7, wz - 3.0);
  placeWanted = false;
  const drift = gy - terrainY(wx, wz);
  setStatus('캐릭터 배치 (' + wx.toFixed(1) + ', ' + gy.toFixed(2) + ', ' + wz.toFixed(1) + ')' +
            (Math.abs(drift) > 0.05 ? ' · 높이맵 오차 ' + drift.toFixed(2) + 'm 보정' : ''));
  if (Math.abs(drift) > 0.05 && hm) hm.bias = (hm.bias || 0) + drift;
}

function applyMapTransform(){
  if (!mapObj) { writeMapOut(); return; }
  const s = mapBaseScale * mapT.s;
  mapObj.scale.setScalar(s);
  mapObj.rotation.y = THREE.MathUtils.degToRad(mapT.rot);
  const off = mapCenter.clone().multiplyScalar(-s).applyAxisAngle(UP, mapObj.rotation.y);
  mapObj.position.set(mapT.x + off.x, -mapMinY*s + mapT.y, mapT.z + off.z);
  writeMapOut();
}

}

function fitMap(target){
  if (!mapObj) return;
  mapObj.scale.setScalar(1); mapObj.position.set(0,0,0); mapObj.rotation.set(0,0,0);
  mapObj.updateMatrixWorld(true);
  const box = new THREE.Box3();
  mapObj.traverse(o => { if (o.isMesh && o.visible && o.parent.visible) box.expandByObject(o); });
  if (box.isEmpty()){ mapBaseScale = 1; mapMinY = 0; refit(); return; }
  const size = box.getSize(new THREE.Vector3());
  const ctr = box.getCenter(new THREE.Vector3());
  mapBaseScale = target / Math.max(size.x, size.z, 1e-6);
  mapMinY = box.min.y;
  mapCenter.set(ctr.x, 0, ctr.z);          // 보이는 파트를 원점으로 끌어온다
  mS.set(1); mYc.set(0); mXc.set(0); mZc.set(0); mRc.set(0);
  refit();
}

// 지형 높이 따라가기 — 구워둔 높이맵을 조회하므로 비용이 사실상 없다
function snapToGround(obj, base, dt){
  if (!obj) return;
  const target = (walkOnMap ? terrainY(obj.position.x, obj.position.z) : 0) + base;
  const d = target - obj.position.y;
  // 낙차가 크면 부드럽게, 작으면 바로 붙인다 (계단식 점프 방지)
  obj.position.y += Math.abs(d) > 0.02 ? d*Math.min(1, dt*16) : d;
}

function nz(i, seed){ const s = Math.sin(i*12.9898 + seed*78.233)*43758.5453; return s - Math.floor(s); }
function duneRing(radius, count, baseH, amp, cNear, cFar, seed){
  const pos = [], col = [];
  const cn = new THREE.Color(cNear), cf = new THREE.Color(cFar), tmp = new THREE.Color();
  const H = i => {
    const k = ((i % count) + count) % count;
    return baseH + amp*(0.35*nz(k, seed) + 0.4*nz(Math.floor(k/3), seed+5) + 0.25*nz(Math.floor(k/9), seed+11));
  };
  const push = (x, y, z, t) => {
    pos.push(x, y, z);
    tmp.copy(cn).lerp(cf, t); col.push(tmp.r, tmp.g, tmp.b);
  };
  for (let i = 0; i < count; i++){
    const a0 = i/count*Math.PI*2, a1 = (i+1)/count*Math.PI*2;
    const h0 = H(i), h1 = H(i+1), top = baseH + amp;
    const x0 = Math.cos(a0)*radius, z0 = Math.sin(a0)*radius;
    const x1 = Math.cos(a1)*radius, z1 = Math.sin(a1)*radius;
    push(x0,-2,z0,0); push(x1,-2,z1,0); push(x1,h1,z1,h1/top);
    push(x0,-2,z0,0); push(x1,h1,z1,h1/top); push(x0,h0,z0,h0/top);
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
  return new THREE.Mesh(g, new THREE.MeshBasicMaterial({ vertexColors:true, fog:true, side:THREE.DoubleSide }));
}

function buildEnv(p){
  while (envGroup.children.length){
    const o = envGroup.children.pop();
    o.traverse(x => { if (x.geometry) x.geometry.dispose(); });
    envGroup.remove(o);
  }
  // 하늘 돔
  skyMat = new THREE.ShaderMaterial({
    side:THREE.BackSide, depthWrite:false, fog:false,
    uniforms:{ top:{value:new THREE.Color(p.top)}, hor:{value:new THREE.Color(p.hor)},
               sunCol:{value:new THREE.Color(p.sun)}, sunDir:{value:new THREE.Vector3(1,0.4,0)},
               haze:{value:envFog} },
    vertexShader:`varying vec3 vD; void main(){ vD = position; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
    fragmentShader:`
      uniform vec3 top, hor, sunCol, sunDir; uniform float haze; varying vec3 vD;
      void main(){
        vec3 d = normalize(vD);
        vec3 c = mix(hor, top, pow(clamp(d.y*1.25, 0.0, 1.0), 0.5));
        float s = max(dot(d, normalize(sunDir)), 0.0);
        c += sunCol*pow(s, 900.0)*4.0;                       // 해 원반
        c += sunCol*pow(s, 6.0)*0.30*(0.4 + haze);           // 주변 글로우
        c = mix(c, hor, pow(clamp(1.0 - abs(d.y)*2.2, 0.0, 1.0), 2.0)*haze*0.7);
        gl_FragColor = vec4(c, 1.0);
      }`
  });
  envGroup.add(new THREE.Mesh(new THREE.SphereGeometry(1600, 32, 20), skyMat));

  // 지평선까지 이어지는 모래 바닥
  const farGround = new THREE.Mesh(new THREE.CircleGeometry(1400, 64),
    new THREE.MeshBasicMaterial({ color:p.sand, fog:true }));
  farGround.rotation.x = -Math.PI/2; farGround.position.y = -0.06;
  envGroup.add(farGround);

  // 사구 두 겹 + 먼 산맥
  envGroup.add(duneRing(150, 90, 4, 12, p.sand, p.hor, 3));
  envGroup.add(duneRing(330, 110, 10, 26, p.rock, p.hor, 9));
  envGroup.add(duneRing(760, 120, 24, 60, p.rock, p.hor, 17));

  // 흩어진 메사 몇 개
  const rockMat = new THREE.MeshLambertMaterial({ color:p.rock, fog:true });
  for (let i = 0; i < 16; i++){
    const ang = nz(i, 31)*Math.PI*2, rad = 70 + nz(i, 47)*180;
    const hgt = 6 + nz(i, 53)*26, rr = 4 + nz(i, 61)*11;
    const m = new THREE.Mesh(new THREE.CylinderGeometry(rr*0.72, rr, hgt, 7, 1), rockMat);
    m.position.set(Math.cos(ang)*rad, hgt/2 - 1, Math.sin(ang)*rad);
    m.rotation.y = nz(i, 71)*3;
    envGroup.add(m);
  }
}

function applyEnv(i){
  envIndex = i;
  [...envListEl.children].forEach((b, n) => b.setAttribute('aria-pressed', String(n === i)));
  if (i === 0){
    envGroup.visible = false; scene.fog = null;
    applyBackground(parseInt(bgPick.value.slice(1), 16));
    hemi.color.setHex(0xffffff); hemi.groundColor.setHex(0x8899aa); hemi.intensity = 1.6;
    key.color.setHex(0xffffff); key.intensity = 2.0; fill.intensity = 0.6;
    ground.visible = !mapObj || !document.getElementById('mGroundChk').checked;
    return;
  }
  const p = ENVS[i];
  buildEnv(p);
  envGroup.visible = true;
  scene.background = new THREE.Color(p.hor);
  scene.fog = new THREE.Fog(p.hor, THREE.MathUtils.lerp(p.fogN*3, p.fogN, envFog), THREE.MathUtils.lerp(p.fogF*2.2, p.fogF, envFog));
  hemi.color.setHex(p.hemiSky); hemi.groundColor.setHex(p.hemiGnd); hemi.intensity = p.hemiI;
  key.color.setHex(p.light); key.intensity = p.keyI;
  fill.color.setHex(p.hemiSky); fill.intensity = 0.35;
  ground.visible = false;
  updateSun();
}
function updateSun(){
  const az = THREE.MathUtils.degToRad(sunAz), el = THREE.MathUtils.degToRad(sunEl);
  const d = new THREE.Vector3(Math.cos(el)*Math.sin(az), Math.sin(el), Math.cos(el)*Math.cos(az));
  key.position.copy(d).multiplyScalar(60);
  if (skyMat) skyMat.uniforms.sunDir.value.copy(d);
}

function addSteppers(){
  document.querySelectorAll('.row input[type=range]').forEach(el => {
    const min = parseFloat(el.min), max = parseFloat(el.max);
    const fine = parseFloat(el.dataset.fine || ((max - min)/200));
    const mk = (txt, dir) => {
      const b = document.createElement('button');
      b.className = 'step'; b.type = 'button'; b.textContent = txt;
      b.tabIndex = -1;
      const go = () => {
        el.value = Math.min(max, Math.max(min, parseFloat(el.value) + dir*fine));
        el.dispatchEvent(new Event('input', { bubbles:true }));
      };
      let hold = null, rep = null;
      b.addEventListener('pointerdown', e => {
        e.preventDefault(); go();
        hold = setTimeout(() => { rep = setInterval(go, 55); }, 380);
      });
      const stop = () => { clearTimeout(hold); clearInterval(rep); };
      ['pointerup','pointerleave','pointercancel'].forEach(ev => b.addEventListener(ev, stop));
      return b;
    };
    el.insertAdjacentElement('beforebegin', mk('−', -1));
    el.insertAdjacentElement('afterend', mk('+', 1));
  });
}
